def norm(x):
    return F.rms_norm(x, (x.size(-1),))


def has_ve(layer_idx, n_layer):
    """Returns True if layer should have Value Embedding (alternating, last always included)."""
    return layer_idx % 2 == (n_layer - 1) % 2


def hash64(key, c1, c2, mask):
    h = (key ^ (key >> 17)) * c1
    h = (h ^ (h >> 23)) * c2
    return (h ^ (h >> 19)) & mask


H_MIX1 = 0x2545F4914F6CDD1D
H_MIX2 = 0x165667B19E3779F9


def mix64(h, tok):
    x = (h ^ (tok + 1)) * H_MIX1
    x = (x ^ (x >> 27)) * H_MIX2
    return x ^ (x >> 31)


def compute_rolling_ctx_hashes(idx, bos_token_id, max_ctx_len=7):
    """Compute rolling 64-bit context hashes and BOS-crossing masks for context lengths 2..max_ctx_len."""
    is_bos = idx == bos_token_id
    invalid = is_bos.clone()
    invalid[:, 0] = True
    h = mix64(torch.zeros_like(idx), idx)
    hashes = []
    invalids = []
    for k in range(2, max_ctx_len + 1):
        prev_h = F.pad(h[:, :-1], (1, 0), value=0)
        h = mix64(prev_h, idx)
        if k > 2:
            invalid = invalid | F.pad(invalid[:, :-1], (1, 0), value=True)
        hashes.append(h)
        invalids.append(invalid)
    return hashes, invalids


class ExactNGramTable(nn.Module):
    """GPU hash table storing exact 32-bit context fingerprints and top-K next-token counts."""

    def __init__(self, num_buckets, k=4):
        super().__init__()
        assert (num_buckets & (num_buckets - 1)) == 0
        self.num_buckets = num_buckets
        self.mask = num_buckets - 1
        self.k = k
        self.register_buffer("keys", torch.zeros(num_buckets, dtype=torch.int32), persistent=False)
        self.register_buffer("total_cnt", torch.zeros(num_buckets, dtype=torch.int32), persistent=False)
        self.register_buffer("cand_tok", torch.full((num_buckets, k), -1, dtype=torch.int16), persistent=False)
        self.register_buffer("cand_cnt", torch.zeros((num_buckets, k), dtype=torch.int32), persistent=False)

    @torch.no_grad()
    def reset(self):
        self.keys.zero_()
        self.total_cnt.zero_()
        self.cand_tok.fill_(-1)
        self.cand_cnt.zero_()

    def query(self, h64, invalid_mask):
        bucket = h64 & self.mask
        fp = ((h64 >> 32).to(torch.int32)) | 1
        cur_fp = self.keys[bucket]
        tot = self.total_cnt[bucket]
        valid = (~invalid_mask) & (cur_fp == fp) & (tot > 0)
        ctok = self.cand_tok[bucket]
        ccnt = self.cand_cnt[bucket]
        tot_f = torch.where(valid, tot, torch.ones_like(tot)).float()
        ccnt_f = torch.where(valid.unsqueeze(-1), ccnt, torch.zeros_like(ccnt)).float()
        cprobs = ccnt_f / tot_f.unsqueeze(-1)
        cov = cprobs.sum(dim=-1).clamp(0.0, 1.0)
        leftover = 1.0 - cov
        top1 = cprobs.amax(dim=-1)
        conc = (cprobs * cprobs).sum(dim=-1)
        logcnt = torch.log1p(torch.where(valid, tot, torch.zeros_like(tot)).float()) * 0.1
        feats = torch.stack([valid.float(), logcnt, top1, cov, conc], dim=-1)
        return valid, ctok, cprobs, leftover, feats

    @torch.no_grad()
    def update(self, h64, targets, invalid_mask):
        valid = ~invalid_mask
        h_v = h64[valid]
        y_v = targets[valid]
        if h_v.numel() == 0:
            return
        bucket = h_v & self.mask
        fp = ((h_v >> 32).to(torch.int32)) | 1

        pair_key = (bucket << 13) | y_v
        uniq_pair, inv, counts = torch.unique(pair_key, return_inverse=True, return_counts=True)
        u_bucket = uniq_pair >> 13
        u_tok = (uniq_pair & 0x1FFF).to(torch.int16)
        u_cnt = counts.to(torch.int32)
        u_fp = torch.empty(uniq_pair.size(0), dtype=torch.int32, device=h64.device)
        u_fp[inv] = fp

        cur_fp = self.keys[u_bucket]
        cur_tot = self.total_cnt[u_bucket]
        is_first = torch.ones(u_bucket.size(0), dtype=torch.bool, device=h64.device)
        if u_bucket.size(0) > 1:
            is_first[1:] = u_bucket[1:] != u_bucket[:-1]
        claim = (cur_fp != u_fp) & ((cur_tot <= 1) | (u_cnt > cur_tot)) & is_first
        if claim.any():
            cb = u_bucket[claim]
            self.keys[cb] = u_fp[claim]
            self.total_cnt[cb] = 0
            self.cand_tok[cb] = -1
            self.cand_cnt[cb] = 0
            cur_fp = self.keys[u_bucket]

        match = cur_fp == u_fp
        if not match.any():
            return
        mb = u_bucket[match]
        mt = u_tok[match]
        mc = u_cnt[match]

        self.total_cnt.index_add_(0, mb, mc)

        ctok = self.cand_tok[mb]
        eq = ctok == mt.unsqueeze(-1)
        hit_any = eq.any(dim=-1)
        if hit_any.any():
            hit_slot = eq[hit_any].int().argmax(dim=-1)
            flat_idx = mb[hit_any] * self.k + hit_slot
            self.cand_cnt.view(-1).index_add_(0, flat_idx, mc[hit_any])

        miss = ~hit_any
        if miss.any():
            miss_b = mb[miss]
            miss_t = mt[miss]
            miss_c = mc[miss]
            first_m = torch.ones(miss_b.size(0), dtype=torch.bool, device=h64.device)
            if miss_b.size(0) > 1:
                first_m[1:] = miss_b[1:] != miss_b[:-1]
            fb = miss_b[first_m]
            ft = miss_t[first_m]
            fc = miss_c[first_m]
            ccnt = self.cand_cnt[fb]
            min_cnt, min_slot = ccnt.min(dim=-1)
            replace = (min_cnt == 0) | (fc > min_cnt)
            if replace.any():
                rb = fb[replace]
                rs = min_slot[replace]
                self.cand_tok[rb, rs] = ft[replace]
                self.cand_cnt[rb, rs] = fc[replace]


def apply_rotary_emb(x, cos, sin):
    assert x.ndim == 4
    d = x.shape[3] // 2
    x1, x2 = x[..., :d], x[..., d:]
    y1 = x1 * cos + x2 * sin
    y2 = x1 * (-sin) + x2 * cos
    return torch.cat([y1, y2], 3)


class CausalSelfAttention(nn.Module):
    def __init__(self, config, layer_idx):
        super().__init__()
        self.n_head = config.n_head
        self.n_kv_head = config.n_kv_head
        self.n_embd = config.n_embd
        self.head_dim = self.n_embd // self.n_head
        assert self.n_embd % self.n_head == 0
        assert self.n_kv_head <= self.n_head and self.n_head % self.n_kv_head == 0
        self.c_q = nn.Linear(self.n_embd, self.n_head * self.head_dim, bias=False)
        self.c_k = nn.Linear(self.n_embd, self.n_kv_head * self.head_dim, bias=False)
        self.c_v = nn.Linear(self.n_embd, self.n_kv_head * self.head_dim, bias=False)
        self.c_proj = nn.Linear(self.n_embd, self.n_embd, bias=False)
        self.ve_gate_channels = 32
        self.ve_gate = nn.Linear(self.ve_gate_channels, self.n_kv_head, bias=False) if has_ve(layer_idx, config.n_layer) else None

    def forward(self, x, ve, cos_sin, window_size):
        B, T, C = x.size()
        q = self.c_q(x).view(B, T, self.n_head, self.head_dim)
        k = self.c_k(x).view(B, T, self.n_kv_head, self.head_dim)
        v = self.c_v(x).view(B, T, self.n_kv_head, self.head_dim)

        # Value residual (ResFormer): mix in value embedding with input-dependent gate per head
        if ve is not None:
            ve = ve.view(B, T, self.n_kv_head, self.head_dim)
            gate = 2 * torch.sigmoid(self.ve_gate(x[..., :self.ve_gate_channels]))
            v = v + gate.unsqueeze(-1) * ve

        cos, sin = cos_sin
        q, k = apply_rotary_emb(q, cos, sin), apply_rotary_emb(k, cos, sin)
        q, k = norm(q), norm(k)

        y = fa3.flash_attn_func(q, k, v, causal=True, window_size=window_size)
        y = y.contiguous().view(B, T, -1)
        y = self.c_proj(y)
        return y


class MLP(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.c_fc = nn.Linear(config.n_embd, 4 * config.n_embd, bias=False)
        self.c_proj = nn.Linear(4 * config.n_embd, config.n_embd, bias=False)

    def forward(self, x):
        x = self.c_fc(x)
        x = F.relu(x).square()
        x = self.c_proj(x)
        return x


class Block(nn.Module):
    def __init__(self, config, layer_idx):
        super().__init__()
        self.attn = CausalSelfAttention(config, layer_idx)
        self.mlp = MLP(config)

    def forward(self, x, ve, cos_sin, window_size):
        x = x + self.attn(norm(x), ve, cos_sin, window_size)
        x = x + self.mlp(norm(x))
        return x


class GPT(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.config = config
        self.window_sizes = self._compute_window_sizes(config)
        self.transformer = nn.ModuleDict({
            "wte": nn.Embedding(config.vocab_size, config.n_embd),
            "h": nn.ModuleList([Block(config, i) for i in range(config.n_layer)]),
        })
        self.lm_head = nn.Linear(config.n_embd, config.vocab_size, bias=False)
        self.resid_lambdas = nn.Parameter(torch.ones(config.n_layer))
        self.x0_lambdas = nn.Parameter(torch.zeros(config.n_layer))
        # Value embeddings
        head_dim = config.n_embd // config.n_head
        kv_dim = config.n_kv_head * head_dim
        self.value_embeds = nn.ModuleDict({
            str(i): nn.Embedding(config.vocab_size, kv_dim)
            for i in range(config.n_layer) if has_ve(i, config.n_layer)
        })
        # Hardware-fused 4-order Input+Output n-gram hash embeddings
        assert config.n_embd % 16 == 0
        assert config.vocab_size <= (1 << 13)
        for sz in (config.bigram_vocab_size, config.trigram_vocab_size,
                   config.fourgram_vocab_size, config.skip_vocab_size):
            assert (sz & (sz - 1)) == 0
        self.bigram_embeds = nn.ModuleList([
            nn.Embedding(config.bigram_vocab_size, config.n_embd),
            nn.Embedding(config.bigram_vocab_size, config.n_embd),
        ])
        self.trigram_embeds = nn.ModuleList([
            nn.Embedding(config.trigram_vocab_size, config.n_embd),
            nn.Embedding(config.trigram_vocab_size, config.n_embd),
        ])
        self.fourgram_dim = 128
        self.fourgram_embeds = nn.ModuleList([
            nn.Embedding(config.fourgram_vocab_size, self.fourgram_dim),
            nn.Embedding(config.fourgram_vocab_size, self.fourgram_dim),
        ])
        self.skip_dim = 256
        self.skip_embeds = nn.ModuleList([
            nn.Embedding(config.skip_vocab_size, self.skip_dim),
            nn.Embedding(config.skip_vocab_size, self.skip_dim),
        ])
        # Nonparametric exact-fingerprint n-gram count tables for orders 3..8 (ctx lengths 2..7)
        exact_bucket_sizes = (8388608, 33554432, 33554432, 16777216, 8388608, 8388608)
        self.num_exact_orders = len(exact_bucket_sizes)
        self.exact_tables = nn.ModuleList([
            ExactNGramTable(sz, k=4) for sz in exact_bucket_sizes
        ])
        # Lightweight projections & 16-group context + Engram key-match output gating + exact mixture gate
        self.out_n_groups = 16
        self.out_key_dim = 8
        self.exact_gate_dim = 64
        self.ngram_proj = nn.ModuleDict({
            "in_extra": nn.Linear(self.fourgram_dim + self.skip_dim, config.n_embd, bias=False),
            "out_4g": nn.Linear(self.fourgram_dim, config.n_embd, bias=False),
            "out_skip": nn.Linear(self.skip_dim, config.n_embd, bias=False),
            "out_ctx": nn.Linear(config.n_embd, 4 * self.out_n_groups, bias=False),
            "out_q": nn.Linear(config.n_embd, self.out_n_groups * self.out_key_dim, bias=False),
            "out_k_bi": nn.Linear(config.n_embd, self.out_n_groups * self.out_key_dim, bias=False),
            "out_k_tri": nn.Linear(config.n_embd, self.out_n_groups * self.out_key_dim, bias=False),
            "out_k_4g": nn.Linear(self.fourgram_dim, self.out_n_groups * self.out_key_dim, bias=False),
            "out_k_skip": nn.Linear(self.skip_dim, self.out_n_groups * self.out_key_dim, bias=False),
            "exact_gate_h": nn.Linear(config.n_embd, self.exact_gate_dim, bias=False),
            "exact_gate_f": nn.Linear(5 * self.num_exact_orders, self.exact_gate_dim, bias=True),
            "exact_gate_out": nn.Linear(self.exact_gate_dim, 1 + self.num_exact_orders, bias=True),
        })
        # Rotary embeddings
        self.rotary_seq_len = config.sequence_len * 10
        cos, sin = self._precompute_rotary_embeddings(self.rotary_seq_len, head_dim)
        self.register_buffer("cos", cos, persistent=False)
        self.register_buffer("sin", sin, persistent=False)

    @torch.no_grad()
    def init_weights(self):
        # Embedding and unembedding
        torch.nn.init.normal_(self.transformer.wte.weight, mean=0.0, std=1.0)
        torch.nn.init.normal_(self.lm_head.weight, mean=0.0, std=0.001)
        # Transformer blocks
        n_embd = self.config.n_embd
        s = 3**0.5 * n_embd**-0.5
        for block in self.transformer.h:
            torch.nn.init.uniform_(block.attn.c_q.weight, -s, s)
            torch.nn.init.uniform_(block.attn.c_k.weight, -s, s)
            torch.nn.init.uniform_(block.attn.c_v.weight, -s, s)
            torch.nn.init.zeros_(block.attn.c_proj.weight)
            torch.nn.init.uniform_(block.mlp.c_fc.weight, -s, s)
            torch.nn.init.zeros_(block.mlp.c_proj.weight)
        # Per-layer scalars
        self.resid_lambdas.fill_(1.0)
        self.x0_lambdas.fill_(0.1)
        # Value embeddings
        for ve in self.value_embeds.values():
            torch.nn.init.uniform_(ve.weight, -s, s)
        # N-gram embeddings init to zero (neutral start)
        for tbl_list in (self.bigram_embeds, self.trigram_embeds, self.fourgram_embeds, self.skip_embeds):
            for tbl in tbl_list:
                torch.nn.init.zeros_(tbl.weight)
        # Reset exact count tables (crucial after to_empty)
        for tbl in self.exact_tables:
            tbl.reset()
        # N-gram projections & gates
        for name, lin in self.ngram_proj.items():
            if name == "out_ctx":
                torch.nn.init.zeros_(lin.weight)
            elif name == "exact_gate_out":
                torch.nn.init.zeros_(lin.weight)
                lin.bias.zero_()
                lin.bias[0] = 2.0
            else:
                s_lin = 3**0.5 * lin.in_features**-0.5
                torch.nn.init.uniform_(lin.weight, -s_lin, s_lin)
                if lin.bias is not None:
                    lin.bias.zero_()
        # Gate weights init to zero (sigmoid(0)=0.5, scaled by 2 -> 1.0 = neutral)
        for block in self.transformer.h:
            if block.attn.ve_gate is not None:
                torch.nn.init.zeros_(block.attn.ve_gate.weight)
        # Rotary embeddings
        head_dim = self.config.n_embd // self.config.n_head
        cos, sin = self._precompute_rotary_embeddings(self.rotary_seq_len, head_dim)
        self.cos, self.sin = cos, sin
        # Cast embeddings to bf16
        self.transformer.wte.to(dtype=torch.bfloat16)
        for ve in self.value_embeds.values():
            ve.to(dtype=torch.bfloat16)
        for tbl_list in (self.bigram_embeds, self.trigram_embeds, self.fourgram_embeds, self.skip_embeds):
            for tbl in tbl_list:
                tbl.to(dtype=torch.bfloat16)

    @torch.no_grad()
    def update_exact_ngrams(self, idx, targets):
        hashes, invalids = compute_rolling_ctx_hashes(idx, self.config.bos_token_id, max_ctx_len=1 + self.num_exact_orders)
        for tbl, h64, inv in zip(self.exact_tables, hashes, invalids):
            tbl.update(h64, targets, inv)

    def _precompute_rotary_embeddings(self, seq_len, head_dim, base=10000, device=None):
        if device is None:
            device = self.transformer.wte.weight.device
        channel_range = torch.arange(0, head_dim, 2, dtype=torch.float32, device=device)
        inv_freq = 1.0 / (base ** (channel_range / head_dim))
        t = torch.arange(seq_len, dtype=torch.float32, device=device)
        freqs = torch.outer(t, inv_freq)
        cos, sin = freqs.cos(), freqs.sin()
        cos, sin = cos.bfloat16(), sin.bfloat16()
        cos, sin = cos[None, :, None, :], sin[None, :, None, :]
        return cos, sin

    def _compute_window_sizes(self, config):
        pattern = config.window_pattern.upper()
        assert all(c in "SL" for c in pattern)
        long_window = config.sequence_len
        short_window = long_window // 2
        char_to_window = {"L": (long_window, 0), "S": (short_window, 0)}
        window_sizes = []
        for layer_idx in range(config.n_layer):
            char = pattern[layer_idx % len(pattern)]
            window_sizes.append(char_to_window[char])
        window_sizes[-1] = (long_window, 0)
        return window_sizes

    def estimate_flops(self):
        """Estimated FLOPs per token (forward + backward)."""
        nparams = sum(p.numel() for p in self.parameters())
        value_embeds_numel = sum(ve.weight.numel() for ve in self.value_embeds.values())
        ngram_embeds_numel = sum(
            tbl.weight.numel()
            for tbl_list in (self.bigram_embeds, self.trigram_embeds, self.fourgram_embeds, self.skip_embeds)
            for tbl in tbl_list
        )
        nparams_exclude = (self.transformer.wte.weight.numel() + value_embeds_numel +
                          ngram_embeds_numel +
                          self.resid_lambdas.numel() + self.x0_lambdas.numel())
        h = self.config.n_head
        q = self.config.n_embd // self.config.n_head
        t = self.config.sequence_len
        attn_flops = 0
        for window_size in self.window_sizes:
            window = window_size[0]
            effective_seq = t if window < 0 else min(window, t)
            attn_flops += 12 * h * q * effective_seq
        return 6 * (nparams - nparams_exclude) + attn_flops

    def num_scaling_params(self):
        wte = sum(p.numel() for p in self.transformer.wte.parameters())
        value_embeds = sum(p.numel() for p in self.value_embeds.parameters())
        bigram_embeds = sum(p.numel() for p in self.bigram_embeds.parameters())
        trigram_embeds = sum(p.numel() for p in self.trigram_embeds.parameters())
        fourgram_embeds = sum(p.numel() for p in self.fourgram_embeds.parameters())
        skip_embeds = sum(p.numel() for p in self.skip_embeds.parameters())
        ngram_proj = sum(p.numel() for p in self.ngram_proj.parameters())
        lm_head = sum(p.numel() for p in self.lm_head.parameters())
        transformer_matrices = sum(p.numel() for p in self.transformer.h.parameters())
        scalars = self.resid_lambdas.numel() + self.x0_lambdas.numel()
        total = (wte + value_embeds + bigram_embeds + trigram_embeds + fourgram_embeds +
                 skip_embeds + ngram_proj + lm_head + transformer_matrices + scalars)
        return {
            'wte': wte, 'value_embeds': value_embeds,
            'bigram_embeds': bigram_embeds, 'trigram_embeds': trigram_embeds,
            'fourgram_embeds': fourgram_embeds, 'skip_embeds': skip_embeds,
            'ngram_proj': ngram_proj,
            'lm_head': lm_head, 'transformer_matrices': transformer_matrices,
            'scalars': scalars, 'total': total,
        }

    def setup_optimizer(self, unembedding_lr=0.004, embedding_lr=0.2, ngram_lr=0.15,
                        matrix_lr=0.02, weight_decay=0.0, adam_betas=(0.8, 0.95), scalar_lr=0.5):
        model_dim = self.config.n_embd
        matrix_params = list(self.transformer.h.parameters())
        value_embeds_params = list(self.value_embeds.parameters())
        ngram_embeds_params = (
            list(self.bigram_embeds.parameters()) +
            list(self.trigram_embeds.parameters()) +
            list(self.fourgram_embeds.parameters()) +
            list(self.skip_embeds.parameters())
        )
        ngram_proj_params = list(self.ngram_proj.parameters())
        embedding_params = list(self.transformer.wte.parameters())
        lm_head_params = list(self.lm_head.parameters())
        resid_params = [self.resid_lambdas]
        x0_params = [self.x0_lambdas]
        assert len(list(self.parameters())) == (len(matrix_params) + len(embedding_params) +
            len(lm_head_params) + len(value_embeds_params) + len(ngram_embeds_params) +
            len(ngram_proj_params) + len(resid_params) + len(x0_params))
        # Scale LR ∝ 1/√dmodel (tuned at 768 dim)
        dmodel_lr_scale = (model_dim / 768) ** -0.5
        print(f"Scaling AdamW LRs by 1/sqrt({model_dim}/768) = {dmodel_lr_scale:.6f}")
        param_groups = [
            dict(kind='adamw', params=lm_head_params, lr=unembedding_lr * dmodel_lr_scale, betas=adam_betas, eps=1e-10, weight_decay=0.0),
            dict(kind='adamw', params=embedding_params, lr=embedding_lr * dmodel_lr_scale, betas=adam_betas, eps=1e-10, weight_decay=0.0),
            dict(kind='adamw', params=value_embeds_params, lr=embedding_lr * dmodel_lr_scale, betas=adam_betas, eps=1e-10, weight_decay=0.0),
            dict(kind='adamw', params=ngram_embeds_params, lr=ngram_lr * dmodel_lr_scale, betas=adam_betas, eps=1e-10, weight_decay=0.0),
            dict(kind='adamw', params=ngram_proj_params, lr=0.01 * dmodel_lr_scale, betas=adam_betas, eps=1e-10, weight_decay=0.0),
            dict(kind='adamw', params=resid_params, lr=scalar_lr * 0.01, betas=adam_betas, eps=1e-10, weight_decay=0.0),
            dict(kind='adamw', params=x0_params, lr=scalar_lr, betas=(0.96, 0.95), eps=1e-10, weight_decay=0.0),
        ]
        for shape in sorted({p.shape for p in matrix_params}):
            group_params = [p for p in matrix_params if p.shape == shape]
            param_groups.append(dict(
                kind='muon', params=group_params, lr=matrix_lr,
                momentum=0.95, ns_steps=5, beta2=0.95, weight_decay=weight_decay,
            ))
        optimizer = MuonAdamW(param_groups)
        for group in optimizer.param_groups:
            group["initial_lr"] = group["lr"]
        return optimizer

    def forward(self, idx, targets=None, reduction='mean'):
        B, T = idx.size()
        assert T <= self.cos.size(1)
        cos_sin = self.cos[:, :T], self.sin[:, :T]

        # Exact bijective n-gram keys masked at BOS document boundaries
        is_bos = idx == self.config.bos_token_id
        prev_idx = F.pad(idx[:, :-1], (1, 0), value=0).masked_fill(is_bos, 0)
        prev_is_bos = F.pad(is_bos[:, :-1], (1, 0), value=True)
        mask2 = is_bos | prev_is_bos
        prev2_idx = F.pad(prev_idx[:, :-1], (1, 0), value=0).masked_fill(mask2, 0)
        prev2_is_bos = F.pad(prev_is_bos[:, :-1], (1, 0), value=True)
        mask3 = mask2 | prev2_is_bos
        prev3_idx = F.pad(prev2_idx[:, :-1], (1, 0), value=0).masked_fill(mask3, 0)

        key2 = (prev_idx << 13) | idx
        key3 = (prev2_idx << 26) | key2
        key4 = (prev3_idx << 39) | key3
        key_skip = (prev2_idx << 13) | idx

        # Two independent 64-bit hash functions per order
        C1A, C2A = 0x2545F4914F6CDD1D, 0x165667B19E3779F9
        C1B, C2B = 0x3C79AC492BA7B653, 0x1C69B3F74AC4AE35

        bmask = self.config.bigram_vocab_size - 1
        bh1 = hash64(key2, C1A, C2A, bmask)
        bh2 = hash64(key2, C1B, C2B, bmask)
        b0 = self.bigram_embeds[0](bh1)
        b1 = self.bigram_embeds[1](bh2)
        b0_in, b0_out = b0.chunk(2, dim=-1)
        b1_in, b1_out = b1.chunk(2, dim=-1)
        bi_in = torch.cat([b0_in, b1_in], dim=-1)
        bi_out = torch.cat([b0_out, b1_out], dim=-1)

        tmask = self.config.trigram_vocab_size - 1
        th1 = hash64(key3, C1A, C2A, tmask)
        th2 = hash64(key3, C1B, C2B, tmask)
        t0 = self.trigram_embeds[0](th1)
        t1 = self.trigram_embeds[1](th2)
        t0_in, t0_out = t0.chunk(2, dim=-1)
        t1_in, t1_out = t1.chunk(2, dim=-1)
        m2 = mask2.unsqueeze(-1)
        tri_in = torch.cat([t0_in, t1_in], dim=-1).masked_fill(m2, 0)
        tri_out = torch.cat([t0_out, t1_out], dim=-1).masked_fill(m2, 0)

        fmask = self.config.fourgram_vocab_size - 1
        fh1 = hash64(key4, C1A, C2A, fmask)
        fh2 = hash64(key4, C1B, C2B, fmask)
        f0 = self.fourgram_embeds[0](fh1)
        f1 = self.fourgram_embeds[1](fh2)
        f0_in, f0_out = f0.chunk(2, dim=-1)
        f1_in, f1_out = f1.chunk(2, dim=-1)
        m3 = mask3.unsqueeze(-1)
        fg_in = torch.cat([f0_in, f1_in], dim=-1).masked_fill(m3, 0)
        fg_out = torch.cat([f0_out, f1_out], dim=-1).masked_fill(m3, 0)

        smask = self.config.skip_vocab_size - 1
        sh1 = hash64(key_skip, C1A, C2A, smask)
        sh2 = hash64(key_skip, C1B, C2B, smask)
        s0 = self.skip_embeds[0](sh1)
        s1 = self.skip_embeds[1](sh2)
        s0_in, s0_out = s0.chunk(2, dim=-1)
        s1_in, s1_out = s1.chunk(2, dim=-1)
        sk_in = torch.cat([s0_in, s1_in], dim=-1).masked_fill(m2, 0)
        sk_out = torch.cat([s0_out, s1_out], dim=-1).masked_fill(m2, 0)

        extra_in = self.ngram_proj["in_extra"](torch.cat([fg_in, sk_in], dim=-1))
        x = self.transformer.wte(idx) + bi_in + tri_in + extra_in
        x = norm(x)
        x0 = x
        for i, block in enumerate(self.transformer.h):
            x = self.resid_lambdas[i] * x + self.x0_lambdas[i] * x0
            ve = self.value_embeds[str(i)](idx) if str(i) in self.value_embeds else None
            x = block(x, ve, cos_sin, self.window_sizes[i])
        h = norm(x)

        # 16-group Context + Engram Key-Match Gated Output N-gram Residual before lm_head
        G = self.out_n_groups
        K = self.out_key_dim
        DG = self.config.n_embd // G
        inv_sqrt_k = K ** -0.5

        ctx = self.ngram_proj["out_ctx"](h).view(B, T, 4, G)
        q_h = F.rms_norm(self.ngram_proj["out_q"](h).view(B, T, G, K), (K,), eps=1e-4)

        def smooth_key_match(proj_k):
            k = proj_k.view(B, T, G, K)
            k = k * torch.rsqrt(k.square().mean(dim=-1, keepdim=True) + 1.0)
            return (q_h * k).sum(dim=-1) * inv_sqrt_k

        g_bi = 2.0 * torch.sigmoid(ctx[:, :, 0, :] + smooth_key_match(self.ngram_proj["out_k_bi"](bi_out)))
        g_tri = 2.0 * torch.sigmoid(ctx[:, :, 1, :] + smooth_key_match(self.ngram_proj["out_k_tri"](tri_out)))
        g_4g = 2.0 * torch.sigmoid(ctx[:, :, 2, :] + smooth_key_match(self.ngram_proj["out_k_4g"](fg_out)))
        g_sk = 2.0 * torch.sigmoid(ctx[:, :, 3, :] + smooth_key_match(self.ngram_proj["out_k_skip"](sk_out)))

        fg_out_full = self.ngram_proj["out_4g"](fg_out)
        sk_out_full = self.ngram_proj["out_skip"](sk_out)

        ngram_out = (
            g_bi.unsqueeze(-1) * bi_out.view(B, T, G, DG) +
            g_tri.unsqueeze(-1) * tri_out.view(B, T, G, DG) +
            g_4g.unsqueeze(-1) * fg_out_full.view(B, T, G, DG) +
            g_sk.unsqueeze(-1) * sk_out_full.view(B, T, G, DG)
        ).view(B, T, self.config.n_embd)

        h = h + ngram_out

        # Query exact-fingerprint nonparametric n-gram count tables (orders 3..8)
        ex_hashes, ex_invalids = compute_rolling_ctx_hashes(idx, self.config.bos_token_id, max_ctx_len=1 + self.num_exact_orders)
        order_valids = []
        order_cands = []
        order_leftovers = []
        order_feats = []
        for tbl, eh, einv in zip(self.exact_tables, ex_hashes, ex_invalids):
            v, ctok, cprobs, leftover, feats = tbl.query(eh, einv)
            order_valids.append(v)
            order_cands.append((v, ctok, cprobs))
            order_leftovers.append(leftover)
            order_feats.append(feats)

        feats_cat = torch.cat(order_feats, dim=-1).to(h.dtype)
        g_hid = F.silu(self.ngram_proj["exact_gate_h"](h) + self.ngram_proj["exact_gate_f"](feats_cat))
        g_logits = self.ngram_proj["exact_gate_out"](g_hid).float()

        valid_all = torch.cat([
            torch.ones((B, T, 1), dtype=torch.bool, device=idx.device),
            torch.stack(order_valids, dim=-1),
        ], dim=-1)
        g_logits = g_logits.masked_fill(~valid_all, -1e4)
        weights = F.softmax(g_logits, dim=-1)
        w_nn = weights[..., 0]
        w_orders = weights[..., 1:]
        leftovers_t = torch.stack(order_leftovers, dim=-1)
        eff_w_nn = w_nn + (w_orders * leftovers_t).sum(dim=-1)

        softcap = 15
        logits = self.lm_head(h)
        logits = logits.float()
        logits = softcap * torch.tanh(logits / softcap)

        if targets is not None:
            log_p_nn_target = -F.cross_entropy(
                logits.view(-1, logits.size(-1)), targets.view(-1),
                ignore_index=-1, reduction='none'
            ).view(B, T)
            t_exp = targets.unsqueeze(-1)
            p_orders_target = torch.stack([
                (cprobs * ((ctok.to(targets.dtype) == t_exp) & v.unsqueeze(-1))).sum(dim=-1)
                for v, ctok, cprobs in order_cands
            ], dim=-1)
            ngram_p_target = (w_orders * p_orders_target).sum(dim=-1)
            log_nn_term = torch.log(eff_w_nn.clamp_min(1e-7)) + log_p_nn_target
            log_p_total = torch.where(
                ngram_p_target > 0,
                torch.logaddexp(log_nn_term, torch.log(ngram_p_target.clamp_min(1e-30))),
                log_nn_term,
            )
            loss = -log_p_total
            if reduction == 'mean':
                return loss.mean()
            elif reduction == 'none':
                return loss.view(-1)
            return loss.sum()

        probs = F.softmax(logits, dim=-1) * eff_w_nn.unsqueeze(-1)
        for o_idx, (v, ctok, cprobs) in enumerate(order_cands):
            add_p = cprobs * (w_orders[..., o_idx] * v.float()).unsqueeze(-1)
            safe_tok = ctok.clamp_min(0).long()
            probs = probs.scatter_add(-1, safe_tok, add_p)
        return torch.log(probs.clamp_min(1e-12))

polar_express_coeffs = [
    (8.156554524902461, -22.48329292557795, 15.878769915207462),
    (4.042929935166739, -2.808917465908714, 0.5000178451051316),
    (3.8916678022926607, -2.772484153217685, 0.5060648178503393),
    (3.285753657755655, -2.3681294933425376, 0.46449024233003106),
    (2.3465413258596377, -1.7097828382687081, 0.42323551169305323),
]

@torch.compile(dynamic=False, fullgraph=True)
def adamw_step_fused(p, grad, exp_avg, exp_avg_sq, step_t, lr_t, beta1_t, beta2_t, eps_t, wd_t):
    p.mul_(1 - lr_t * wd_t)
    exp_avg.lerp_(grad, 1 - beta1_t)
    exp_avg_sq.lerp_(grad.square(), 1 - beta2_t)
    bias1 = 1 - beta1_t ** step_t
    bias2 = 1 - beta2_t ** step_t
    denom = (exp_avg_sq / bias2).sqrt() + eps_t
    step_size = lr_t / bias1
    p.add_(exp_avg / denom, alpha=-step_size)

@torch.compile(dynamic=False, fullgraph=True)
def muon_step_fused(stacked_grads, stacked_params, momentum_buffer, second_momentum_buffer,
                    momentum_t, lr_t, wd_t, beta2_t, ns_steps, red_dim):
    # Nesterov momentum
    momentum = momentum_t.to(stacked_grads.dtype)
    momentum_buffer.lerp_(stacked_grads, 1 - momentum)
    g = stacked_grads.lerp_(momentum_buffer, momentum)
    # Polar express orthogonalization
    X = g.bfloat16()
    X = X / (X.norm(dim=(-2, -1), keepdim=True) * 1.02 + 1e-6)
    if g.size(-2) > g.size(-1):
        for a, b, c in polar_express_coeffs[:ns_steps]:
            A = X.mT @ X
            B = b * A + c * (A @ A)
            X = a * X + X @ B
    else:
        for a, b, c in polar_express_coeffs[:ns_steps]:
            A = X @ X.mT
            B = b * A + c * (A @ A)
            X = a * X + B @ X
    g = X
    # NorMuon variance reduction
    beta2 = beta2_t.to(g.dtype)
    v_mean = g.float().square().mean(dim=red_dim, keepdim=True)
    red_dim_size = g.size(red_dim)
    v_norm_sq = v_mean.sum(dim=(-2, -1), keepdim=True) * red_dim_size
    v_norm = v_norm_sq.sqrt()
    second_momentum_buffer.lerp_(v_mean.to(dtype=second_momentum_buffer.dtype), 1 - beta2)
    step_size = second_momentum_buffer.clamp_min(1e-10).rsqrt()
    scaled_sq_sum = (v_mean * red_dim_size) * step_size.float().square()
    v_norm_new = scaled_sq_sum.sum(dim=(-2, -1), keepdim=True).sqrt()
    final_scale = step_size * (v_norm / v_norm_new.clamp_min(1e-10))
    g = g * final_scale.to(g.dtype)
    # Cautious weight decay + parameter update
    lr = lr_t.to(g.dtype)
    wd = wd_t.to(g.dtype)
    mask = (g * stacked_params) >= 0
    stacked_params.sub_(lr * g + lr * wd * stacked_params * mask)


class MuonAdamW(torch.optim.Optimizer):
    """Combined optimizer: Muon for 2D matrix params, AdamW for others."""

    def __init__(self, param_groups):
        super().__init__(param_groups, defaults={})
        # 0-D CPU tensors to avoid torch.compile recompilation when values change
        self._adamw_step_t = torch.tensor(0.0, dtype=torch.float32, device="cpu")
        self._adamw_lr_t = torch.tensor(0.0, dtype=torch.float32, device="cpu")
        self._adamw_beta1_t = torch.tensor(0.0, dtype=torch.float32, device="cpu")
        self._adamw_beta2_t = torch.tensor(0.0, dtype=torch.float32, device="cpu")
        self._adamw_eps_t = torch.tensor(0.0, dtype=torch.float32, device="cpu")
        self._adamw_wd_t = torch.tensor(0.0, dtype=torch.float32, device="cpu")
        self._muon_momentum_t = torch.tensor(0.0, dtype=torch.float32, device="cpu")
        self._muon_lr_t = torch.tensor(0.0, dtype=torch.float32, device="cpu")
        self._muon_wd_t = torch.tensor(0.0, dtype=torch.float32, device="cpu")
        self._muon_beta2_t = torch.tensor(0.0, dtype=torch.float32, device="cpu")

    def _step_adamw(self, group):
        for p in group['params']:
            if p.grad is None:
                continue
            grad = p.grad
            state = self.state[p]
            if not state:
                state['step'] = 0
                state['exp_avg'] = torch.zeros_like(p)
                state['exp_avg_sq'] = torch.zeros_like(p)
            state['step'] += 1
            self._adamw_step_t.fill_(state['step'])
            self._adamw_lr_t.fill_(group['lr'])
            self._adamw_beta1_t.fill_(group['betas'][0])
            self._adamw_beta2_t.fill_(group['betas'][1])
            self._adamw_eps_t.fill_(group['eps'])
            self._adamw_wd_t.fill_(group['weight_decay'])
            adamw_step_fused(p, grad, state['exp_avg'], state['exp_avg_sq'],
                            self._adamw_step_t, self._adamw_lr_t, self._adamw_beta1_t,
                            self._adamw_beta2_t, self._adamw_eps_t, self._adamw_wd_t)

    def _step_muon(self, group):
        params = group['params']
        if not params:
            return
        p = params[0]
        state = self.state[p]
        num_params = len(params)
        shape, device, dtype = p.shape, p.device, p.dtype
        if "momentum_buffer" not in state:
            state["momentum_buffer"] = torch.zeros(num_params, *shape, dtype=dtype, device=device)
        if "second_momentum_buffer" not in state:
            state_shape = (num_params, shape[-2], 1) if shape[-2] >= shape
