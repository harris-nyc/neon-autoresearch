# Fused Chunked LM Head + Softcap + Cross-Entropy
# ---------------------------------------------------------------------------

@torch.compile(dynamic=False, fullgraph=True)
def _softcap_ce_chunk_fwd_bwd(raw_bf16, y_c, softcap, inv_n):
    raw = raw_bf16.float()
    t = torch.tanh(raw / softcap)
    logits = softcap * t
    lse = torch.logsumexp(logits, dim=-1, keepdim=True)
    nll = lse.squeeze(-1) - logits.gather(-1, y_c.unsqueeze(-1)).squeeze(-1)
    loss_sum = nll.sum()
    probs = torch.exp(logits - lse)
    neg_ones = torch.full((y_c.size(0), 1), -1.0, device=probs.device, dtype=probs.dtype)
    d_logits = probs.scatter_add(-1, y_c.unsqueeze(-1), neg_ones)
    d_raw = (d_logits * (1.0 - t * t) * inv_n).to(raw_bf16.dtype)
    return loss_sum, d_raw


@torch.compile(dynamic=False, fullgraph=True)
def _softcap_ce_chunk_eval(raw_bf16, y_c, softcap):
    logits = softcap * torch.tanh(raw_bf16.float() / softcap)
    return F.cross_entropy(logits, y_c, ignore_index=-1, reduction='none')


class FusedChunkedLMHead(torch.autograd.Function):
    @staticmethod
    def forward(ctx, h, weight, targets, softcap):
        B, T, D = h.shape
        N = B * T
        h_flat = h.reshape(N, D)
        y_flat = targets.reshape(N)
        w_bf16 = weight.to(h_flat.dtype)
        d_h = torch.empty_like(h_flat)
        d_w = torch.zeros_like(weight, dtype=torch.float32)
        loss_sum = torch.zeros((), device=h.device, dtype=torch.float32)
        inv_n = 1.0 / N
        chunk_size = 32768
        for start in range(0, N, chunk_size):
            end = min(start + chunk_size, N)
            h_c = h_flat[start:end]
            y_c = y_flat[start:end]
            raw_c = F.linear(h_c, w_bf16)
            l_c, d_raw_c = _softcap_ce_chunk_fwd_bwd(raw_c, y_c, softcap, inv_n)
            loss_sum = loss_sum + l_c
            d_h[start:end] = d_raw_c @ w_bf16
            d_w.add_((d_raw_c.mT @ h_c).float())
        ctx.save_for_backward(d_h.view(B, T, D), d_w.to(weight.dtype))
        return loss_sum * inv_n

    @staticmethod
    def backward(ctx, grad_output):
        d_h, d_w = ctx.saved_tensors
        return d_h * grad_output.to(d_h.dtype), d_w * grad_output.to(d_w.dtype), None, None

def norm(x):
    return F.rms_norm(x, (x.size(-1),))


def has_ve(layer_idx, n_layer):
    """Returns True if layer should have Value Embedding (alternating, last always included)."""
    return layer_idx % 2 == (n_layer - 1) % 2


def hash64(key, c1, c2, mask):
    h = (key ^ (key >> 17)) * c1
    h = (h ^ (h >> 23)) * c2
    return (h ^ (h >> 19)) & mask


def apply_rotary_emb(x, cos, sin):
    assert x.ndim == 4
    d = x.shape[3] // 2
    x1, x2 = x[..., :d], x[..., d:]
    y1 = x1 * cos + x2 * sin
    y2 = x1 * (-sin) + x2 * cos
    return torch.cat([y1, y2], 3)


class CausalSelfAttention(nn.Module):
    def __init__(self, config, layer_idx):
        super().__init__()
        self.n_head = config.n_head
        self.n_kv_head = config.n_kv_head
        self.n_embd = config.n_embd
        self.head_dim = self.n_embd // self.n_head
        assert self.n_embd % self.n_head == 0
        assert self.n_kv_head <= self.n_head and self.n_head % self.n_kv_head == 0
        self.c_q = nn.Linear(self.n_embd, self.n_head * self.head_dim, bias=False)
        self.c_k = nn.Linear(self.n_embd, self.n_kv_head * self.head_dim, bias=False)
        self.c_v = nn.Linear(self.n_embd, self.n_kv_head * self.head_dim, bias=False)
        self.c_proj = nn.Linear(self.n_embd, self.n_embd, bias=False)
        self.ve_gate_channels = 32
        self.ve_gate = nn.Linear(self.ve_gate_channels, self.n_kv_head, bias=False) if has_ve(layer_idx, config.n_layer) else None

    def forward(self, x, ve, cos_sin, window_size):
        B, T, C = x.size()
        q = self.c_q(x).view(B, T, self.n_head, self.head_dim)
        k = self.c_k(x).view(B, T, self.n_kv_head, self.head_dim)
        v = self.c_v(x).view(B, T, self.n_kv_head, self.head_dim)

        # Value residual (ResFormer): mix in value embedding with input-dependent gate per head
        if ve is not None:
            ve = ve.view(B, T, self.n_kv_head, self.head_dim)
            gate = 2 * torch.sigmoid(self.ve_gate(x[..., :self.ve_gate_channels]))
            v = v + gate.unsqueeze(-1) * ve

        cos, sin = cos_sin
        q, k = apply_rotary_emb(q, cos, sin), apply_rotary_emb(k, cos, sin)
        q, k = norm(q), norm(k)

        y = fa3.flash_attn_func(q, k, v, causal=True, window_size=window_size)
        y = y.contiguous().view(B, T, -1)
        y = self.c_proj(y)
        return y


class MLP(nn.Module):
    def __init__(self, config, mult=4):
        super().__init__()
        hidden = mult * config.n_embd
        self.c_fc = nn.Linear(config.n_embd, hidden, bias=False)
        self.c_proj = nn.Linear(hidden, config.n_embd, bias=False)

    def forward(self, x):
        x = self.c_fc(x)
        x = F.relu(x).square()
        x = self.c_proj(x)
        return x


class Block(nn.Module):
    def __init__(self, config, layer_idx):
        super().__init__()
        self.attn = CausalSelfAttention(config, layer_idx)
        self.mlp = MLP(config, mult=4)

    def forward(self, x, ve, cos_sin, window_size):
        x = x + self.attn(norm(x), ve, cos_sin, window_size)
        x = x + self.mlp(norm(x))
        return x


class GPT(nn.Module):
    def __init__(self, config):
        super().__init__()
        self.config = config
        self.window_sizes = self._compute_window_sizes(config)
        self.transformer = nn.ModuleDict({
            "wte": nn.Embedding(config.vocab_size, config.n_embd),
            "h": nn.ModuleList([Block(config, i) for i in range(config.n_layer)]),
        })
        self.lm_head = nn.Linear(config.n_embd, config.vocab_size, bias=False)
        self.resid_lambdas = nn.Parameter(torch.ones(config.n_layer))
        self.x0_lambdas = nn.Parameter(torch.zeros(config.n_layer))
        # Value embeddings
        head_dim = config.n_embd // config.n_head
        kv_dim = config.n_kv_head * head_dim
        self.value_embeds = nn.ModuleDict({
            str(i): nn.Embedding(config.vocab_size, kv_dim)
            for i in range(config.n_layer) if has_ve(i, config.n_layer)
        })
        # Collision-optimal 4-order Input+Output n-gram hash embeddings (exact same 1.61B params,
        # 2x-4x more buckets, 2.44x less gather/scatter bandwidth)
        # Each head row stores [input_part | output_part] contiguously.
        assert config.n_embd % 16 == 0
        assert config.vocab_size <= (1 << 13)
        for sz in (config.bigram_vocab_size, config.trigram_vocab_size,
                   config.fourgram_vocab_size, config.skip_vocab_size):
            assert (sz & (sz - 1)) == 0
        # Bigram: 2 heads x 1M x (128 in + 128 out = 256) -> 256 in, 256 out
        self.bigram_dim = 256
        self.bigram_embeds = nn.ModuleList([
            nn.Embedding(config.bigram_vocab_size, self.bigram_dim),
            nn.Embedding(config.bigram_vocab_size, self.bigram_dim),
        ])
        # Trigram: 2 heads x 2M x (64 in + 64 out = 128) -> 128 in, 128 out
        self.trigram_dim = 128
        self.trigram_embeds = nn.ModuleList([
            nn.Embedding(config.trigram_vocab_size, self.trigram_dim),
            nn.Embedding(config.trigram_vocab_size, self.trigram_dim),
        ])
        # 4-gram: 2 heads x 2M x (32 in + 32 out = 64) -> 64 in, 64 out
        self.fourgram_dim = 64
        self.fourgram_embeds = nn.ModuleList([
            nn.Embedding(config.fourgram_vocab_size, self.fourgram_dim),
            nn.Embedding(config.fourgram_vocab_size, self.fourgram_dim),
        ])
        # Skip-bigram (t-2, t): 2 heads x 1M x (64 in + 64 out = 128) -> 128 in, 128 out
        self.skip_dim = 128
        self.skip_embeds = nn.ModuleList([
            nn.Embedding(config.skip_vocab_size, self.skip_dim),
            nn.Embedding(config.skip_vocab_size, self.skip_dim),
        ])
        self.ngram_total_dim = self.bigram_dim + self.trigram_dim + self.fourgram_dim + self.skip_dim  # 576

        # 2D N-gram projection matrices (Muon-optimized) + nonlinear output fusion MLP
        self.ngram_lin = nn.ModuleDict({
            "in_proj": nn.Linear(self.ngram_total_dim, config.n_embd, bias=False),
            "mid_proj": nn.Linear(self.ngram_total_dim, config.n_embd, bias=False),
            "out_proj": nn.Linear(self.ngram_total_dim, config.n_embd, bias=False),
        })
        self.out_mlp = MLP(config, mult=2)

        # Strictly independent per-head key-match + context gating (8 groups per head x 2 heads = 16 groups)
        self.n_groups = 16
        self.n_groups_per_head = 8
        self.key_dim = 8
        gk_head = self.n_groups_per_head * self.key_dim  # 64
        gk_all = self.n_groups * self.key_dim            # 128
        self.ngram_gates = nn.ModuleDict({
            # Mid-network (after Block 1) collision-filtering gates on input slices
            "mid_ctx": nn.Linear(config.n_embd, 4 * self.n_groups, bias=False),
            "mid_q": nn.Linear(config.n_embd, gk_all, bias=False),
            "mid_k_bi0": nn.Linear(self.bigram_dim // 2, gk_head, bias=False),
            "mid_k_bi1": nn.Linear(self.bigram_dim // 2, gk_head, bias=False),
            "mid_k_tri0": nn.Linear(self.trigram_dim // 2, gk_head, bias=False),
            "mid_k_tri1": nn.Linear(self.trigram_dim // 2, gk_head, bias=False),
            "mid_k_4g0": nn.Linear(self.fourgram_dim // 2, gk_head, bias=False),
            "mid_k_4g1": nn.Linear(self.fourgram_dim // 2, gk_head, bias=False),
            "mid_k_sk0": nn.Linear(self.skip_dim // 2, gk_head, bias=False),
            "mid_k_sk1": nn.Linear(self.skip_dim // 2, gk_head, bias=False),
            # Output (after Block 7) collision-filtering gates on output slices
            "out_ctx": nn.Linear(config.n_embd, 4 * self.n_groups, bias=False),
            "out_q": nn.Linear(config.n_embd, gk_all, bias=False),
            "out_k_bi0": nn.Linear(self.bigram_dim // 2, gk_head, bias=False),
            "out_k_bi1": nn.Linear(self.bigram_dim // 2, gk_head, bias=False),
            "out_k_tri0": nn.Linear(self.trigram_dim // 2, gk_head, bias=False),
            "out_k_tri1": nn.Linear(self.trigram_dim // 2, gk_head, bias=False),
            "out_k_4g0": nn.Linear(self.fourgram_dim // 2, gk_head, bias=False),
            "out_k_4g1": nn.Linear(self.fourgram_dim // 2, gk_head, bias=False),
            "out_k_sk0": nn.Linear(self.skip_dim // 2, gk_head, bias=False),
            "out_k_sk1": nn.Linear(self.skip_dim // 2, gk_head, bias=False),
        })

        # Rotary embeddings
        self.rotary_seq_len = config.sequence_len * 10
        cos, sin = self._precompute_rotary_embeddings(self.rotary_seq_len, head_dim)
        self.register_buffer("cos", cos, persistent=False)
        self.register_buffer("sin", sin, persistent=False)
        self._compiled_backbone = None

    @torch.no_grad()
    def init_weights(self):
        # Embedding and unembedding
        torch.nn.init.normal_(self.transformer.wte.weight, mean=0.0, std=1.0)
        torch.nn.init.normal_(self.lm_head.weight, mean=0.0, std=0.001)
        # Transformer blocks
        n_embd = self.config.n_embd
        s = 3**0.5 * n_embd**-0.5
        for block in self.transformer.h:
            torch.nn.init.uniform_(block.attn.c_q.weight, -s, s)
            torch.nn.init.uniform_(block.attn.c_k.weight, -s, s)
            torch.nn.init.uniform_(block.attn.c_v.weight, -s, s)
            torch.nn.init.zeros_(block.attn.c_proj.weight)
            torch.nn.init.uniform_(block.mlp.c_fc.weight, -s, s)
            torch.nn.init.zeros_(block.mlp.c_proj.weight)
        # Output fusion MLP (zero-init c_proj -> neutral start)
        torch.nn.init.uniform_(self.out_mlp.c_fc.weight, -s, s)
        torch.nn.init.zeros_(self.out_mlp.c_proj.weight)
        # Per-layer scalars
        self.resid_lambdas.fill_(1.0)
        self.x0_lambdas.fill_(0.1)
        # Value embeddings
        for ve in self.value_embeds.values():
            torch.nn.init.uniform_(ve.weight, -s, s)
        # N-gram embeddings init to zero (neutral start)
        for tbl_list in (self.bigram_embeds, self.trigram_embeds, self.fourgram_embeds, self.skip_embeds):
            for tbl in tbl_list:
                torch.nn.init.zeros_(tbl.weight)
        # N-gram linear projections
        for name, lin in self.ngram_lin.items():
            s_lin = 3**0.5 * lin.in_features**-0.5
            if name == "mid_proj":
                torch.nn.init.uniform_(lin.weight, -0.1 * s_lin, 0.1 * s_lin)
            else:
                torch.nn.init.uniform_(lin.weight, -s_lin, s_lin)
        # N-gram gates
        for name, lin in self.ngram_gates.items():
            if name.endswith("_ctx"):
                torch.nn.init.zeros_(lin.weight)
            else:
                s_lin = 3**0.5 * lin.in_features**-0.5
                torch.nn.init.uniform_(lin.weight, -s_lin, s_lin)
        # Gate weights init to zero (sigmoid(0)=0.5, scaled by 2 -> 1.0 = neutral)
        for block in self.transformer.h:
            if block.attn.ve_gate is not None:
                torch.nn.init.zeros_(block.attn.ve_gate.weight)
        # Rotary embeddings
        head_dim = self.config.n_embd // self.config.n_head
        cos, sin = self._precompute_rotary_embeddings(self.rotary_seq_len, head_dim)
        self.cos, self.sin = cos, sin
        # Cast embeddings to bf16
        self.transformer.wte.to(dtype=torch.bfloat16)
        for ve in self.value_embeds.values():
            ve.to(dtype=torch.bfloat16)
        for tbl_list in (self.bigram_embeds, self.trigram_embeds, self.fourgram_embeds, self.skip_embeds):
            for tbl in tbl_list:
                tbl.to(dtype=torch.bfloat16)
